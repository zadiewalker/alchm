  'suicide', 'kill myself', 'end it all', 'no point living',
  'hurt myself', 'can\'t go on', 'want to die'
];

function detectCrisis(journalEntry) {
  const hasCrisisLanguage = crisisKeywords.some(keyword => 
    journalEntry.toLowerCase().includes(keyword)
  );

    logCrisisEvent();
